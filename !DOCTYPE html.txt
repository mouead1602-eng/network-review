<script>
    // Quiz Questions Data
    const questions = [
        {
            category: "شبكات الحاسوب",
            question: "أي عبارة تمثل تعريف شبكة الحاسوب؟",
            options: [
                "حاسوب واحد يعمل دون اتصال",
                "حواسيب متصلة لمشاركة البيانات والأجهزة",
                "برنامج لحماية الجهاز فقط",
                "وسيلة لتخزين الصور فقط"
            ],
            correct: 1
        },
        {
            category: "مخططات التوصيل",
            question: "ما هو مخطط التوصيل الأكثر شيوعاً والذي يعتمد على وجود جهاز مركزي؟",
            options: [
                "المخطط الخطي (Bus)",
                "المخطط الحلقي (Ring)",
                "المخطط النجمي (Star)",
                "المخطط المتشابك (Mesh)"
            ],
            correct: 2
        },
        {
            category: "البروتوكولات",
            question: "كم عدد طبقات نموذج OSI للشبكات؟",
            options: [
                "4 طبقات",
                "5 طبقات",
                "7 طبقات",
                "3 طبقات"
            ],
            correct: 2
        }
    ];

    let currentIndex = 0;
    let xp = 0;
    let correctCount = 0;
    let userAnswers = new Array(questions.length).fill(null);

    // Initialize App
    function initQuiz() {
        renderQuestion();
        renderQuickNav();
    }

    function renderQuestion() {
        const q = questions[currentIndex];
        document.getElementById('category-tag').innerText = q.category;
        document.getElementById('question-count').innerText = `سؤال ${currentIndex + 1} من ${questions.length}`;
        document.getElementById('question-text').innerText = q.question;
        document.getElementById('total-q-count').innerText = questions.length;

        const optionsContainer = document.getElementById('options-container');
        optionsContainer.innerHTML = '';

        q.options.forEach((opt, idx) => {
            const btn = document.createElement('button');
            btn.className = 'option-btn';
            btn.innerText = opt;
            
            // Check if already answered
            if(userAnswers[currentIndex] !== null) {
                if(idx === q.correct) btn.classList.add('correct');
                if(userAnswers[currentIndex] === idx && idx !== q.correct) btn.classList.add('wrong');
            } else {
                btn.onclick = () => selectOption(idx);
            }
            
            optionsContainer.appendChild(btn);
        });
    }

    function selectOption(index) {
        if(userAnswers[currentIndex] !== null) return; // Prevent re-answering

        userAnswers[currentIndex] = index;
        const q = questions[currentIndex];

        if(index === q.correct) {
            xp += 10;
            correctCount++;
        }

        updateStats();
        renderQuestion();
        renderQuickNav();
    }

    function updateStats() {
        document.getElementById('xp-val').innerText = xp;
        document.getElementById('correct-val').innerText = correctCount;
        
        const answeredCount = userAnswers.filter(a => a !== null).length;
        const ratio = answeredCount > 0 ? Math.round((correctCount / questions.length) * 100) : 0;
        document.getElementById('ratio-val').innerText = ratio + '%';
        
        document.getElementById('badge-val').innerText = Math.floor(xp / 20);
    }

    function renderQuickNav() {
        const navContainer = document.getElementById('quick-nav');
        navContainer.innerHTML = '';

        questions.forEach((_, idx) => {
            const div = document.createElement('div');
            div.className = `nav-num ${currentIndex === idx ? 'active' : ''} ${userAnswers[idx] !== null ? 'answered' : ''}`;
            div.innerText = idx + 1;
            div.onclick = () => {
                currentIndex = idx;
                renderQuestion();
                renderQuickNav();
            };
            navContainer.appendChild(div);
        });
    }

    function nextQuestion() {
        if (currentIndex < questions.length - 1) {
            currentIndex++;
            renderQuestion();
            renderQuickNav();
        }
    }

    function prevQuestion() {
        if (currentIndex > 0) {
            currentIndex--;
            renderQuestion();
            renderQuickNav();
        }
    }

    // Start App
    initQuiz();
</script>
